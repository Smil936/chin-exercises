🔧 Google Sheet Setup Guide
One-time setup (5 minutes):

1. Go to sheets.new and create a new spreadsheet
2. Click Extensions → Apps Script
3. Delete existing code and paste the script below
4. Click Deploy → New deployment → Web app
5. Set "Who has access" to Anyone
6. Copy the Web App URL and paste it above

Apps Script to paste:
function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // Add header if sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow(['Date', 'Exercises Done', 'Total', 'Completed?', 'Exercises List']); } // Update today's row if it exists, otherwise append var values = sheet.getDataRange().getValues(); var todayRow = -1; for (var i = 1; i < values.length; i++) { if (values[i][0] === data.date) { todayRow = i + 1; break; } } var row = [data.date, data.count, data.total, data.completed, data.exercises]; if (todayRow > 0) { sheet.getRange(todayRow, 1, 1, 5).setValues([row]); } else { sheet.appendRow(row); } return ContentService .createTextOutput(JSON.stringify({status: 'ok'})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({status: 'ok', message: 'Chin Exercise Logger'})) .setMimeType(ContentService.MimeType.JSON); }
